cmake_minimum_required(VERSION 3.13)

# Builds the SWIG C# bindings for libquicr (from ../../../../swig/quicr.i)
# and a tiny C# program that exercises them - the C# counterpart to ../go
# and ../python, driven by CMake the same way so that `cmake --build`
# alone reproduces what running swig/csharp by hand does.
#
# Both SWIG and a .NET SDK are optional dependencies of the rest of the
# project, so this whole example is skipped (with a message, not an
# error) if either is missing.

find_package(SWIG 4.0 QUIET)
find_program(DOTNET_EXECUTABLE dotnet)

if(NOT SWIG_FOUND OR NOT DOTNET_EXECUTABLE)
    message(STATUS "Skipping SWIG C# example (qcsclient): SWIG and/or the .NET SDK were not found")
    return()
endif()

cmake_policy(SET CMP0086 NEW)
if(POLICY CMP0122)
    # NEW: UseSWIG names the csharp module target using normal CMake shared-
    # library conventions (respects OUTPUT_NAME below) rather than its own
    # legacy SWIG-specific naming scheme.
    cmake_policy(SET CMP0122 NEW)
endif()
include(UseSWIG)

# Directory layout under the build tree:
#   pkg/   the generated C# proxy classes (quicrPINVOKE.cs, ObjectHeaders.cs,
#          SessionManager.cs, ...) plus the native libquicr.{dylib,so} P/Invoke
#          calls into
#   app/   a generated qcsclient.csproj (referencing pkg/ by its real build-
#          tree path - the C# equivalent of the Go example's generated
#          go.mod `replace` directive, needed for the same reason: pkg/'s
#          path is only known at CMake configure time) plus a copy of
#          Program.cs, and `dotnet build`'s own output (app/out/)
set(QUICR_CS_PKG_DIR ${CMAKE_CURRENT_BINARY_DIR}/pkg)
set(QUICR_CS_APP_DIR ${CMAKE_CURRENT_BINARY_DIR}/app)
set(QUICR_CS_OUT_DIR ${QUICR_CS_APP_DIR}/out)
file(MAKE_DIRECTORY ${QUICR_CS_PKG_DIR})
file(MAKE_DIRECTORY ${QUICR_CS_APP_DIR})

set_property(SOURCE ${PROJECT_SOURCE_DIR}/swig/quicr.i PROPERTY CPLUSPLUS ON)
# Without -namespace, SWIG emits every generated class at global (no
# namespace) scope - Program.cs's own `using quicr;` (matching the Go/
# Python examples' own `import quicr`/`"quicr"` package-qualified access)
# needs this to actually mean something.
set(CMAKE_SWIG_FLAGS -namespace quicr)
set(CMAKE_SWIG_OUTDIR ${QUICR_CS_PKG_DIR})

# CMAKE_SWIG_OUTDIR only controls where the generated .cs proxy files go;
# SWIG_OUTFILE_DIR (defaulting to CMAKE_SWIG_OUTDIR if unset) governs the
# generated wrap .cxx - kept out of pkg/ the same way the Go/Python
# examples keep it out of their own generated-package directories.
set(SWIG_OUTFILE_DIR ${CMAKE_CURRENT_BINARY_DIR}/wrap)

# UseSWIG derives the `-I...` flags it passes to the swig executable itself
# (as opposed to the flags used to *compile* the generated wrapper) from
# this directory-scoped property, not from target_include_directories().
include_directories(${CMAKE_BINARY_DIR}/include ${PROJECT_SOURCE_DIR}/include)

swig_add_library(quicr_csharp
    LANGUAGE csharp
    TYPE SHARED
    SOURCES ${PROJECT_SOURCE_DIR}/swig/quicr.i
)

set_target_properties(quicr_csharp PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED YES
    CXX_EXTENSIONS OFF
    LIBRARY_OUTPUT_DIRECTORY ${QUICR_CS_PKG_DIR}
    RUNTIME_OUTPUT_DIRECTORY ${QUICR_CS_PKG_DIR}
    # `[DllImport("quicr")]` (the generated quicrPINVOKE.cs's own default,
    # taken from `%module quicr` in quicr.i - see csharp/typemaps.i's own
    # DllImport-name comments) needs the built library to actually be
    # named libquicr.{dylib,so}/quicr.dll, not libquicr_csharp.*; matches
    # the Python example's own identical OUTPUT_NAME "quicr" override, for
    # the same reason (there, `import quicr` needing `_quicr.*.so`).
    OUTPUT_NAME "quicr"
)

# quicr_csharp only needs to link quicr itself: because it's a real CMake
# target (unlike a manual swig/g++ invocation), CMake resolves every one of
# quicr's transitive PUBLIC dependencies (spdlog, timeq, picoquic, picotls,
# sframe, OpenSSL/mbedTLS, fmt, ...) automatically, the same as the Go/
# Python examples' own targets.
target_link_libraries(quicr_csharp PRIVATE quicr)

if(LINT)
    include(Lint)
    lint(quicr_csharp)
endif()

configure_file(Program.cs ${QUICR_CS_APP_DIR}/Program.cs COPYONLY)

# Written with file(GENERATE) rather than checked into source because it
# must embed QUICR_CS_PKG_DIR's real build-tree path - the same reason the
# Go example's own go.mod/cgo_link.go are generated rather than checked in.
# EnableDefaultCompileItems is off because pkg/'s generated .cs files live
# outside app/'s own directory tree (SDK-style default globs only look
# under the project's own directory), so they're listed explicitly instead.
file(GENERATE
    OUTPUT ${QUICR_CS_APP_DIR}/qcsclient.csproj
    CONTENT "<!-- Code generated by CMake at configure time. DO NOT EDIT. -->
<Project Sdk=\"Microsoft.NET.Sdk\">
  <PropertyGroup>
    <OutputType>Exe</OutputType>
    <TargetFramework>net8.0</TargetFramework>
    <ImplicitUsings>enable</ImplicitUsings>
    <Nullable>enable</Nullable>
    <AssemblyName>qcsclient</AssemblyName>
    <RootNamespace>qcsclient</RootNamespace>
    <EnableDefaultCompileItems>false</EnableDefaultCompileItems>
    <InvariantGlobalization>true</InvariantGlobalization>
    <!-- net8.0 is the baseline TFM to build against, but only whichever
         .NET *runtime* happens to be installed actually runs it - rolling
         forward to that (rather than requiring an exact 8.0.x runtime
         alongside newer ones, e.g. 10.0.x here) avoids a real, reproduced
         \"You must install or update .NET to run this application\"
         launch failure on any machine whose only installed runtime is
         newer than 8.0. -->
    <RollForward>LatestMajor</RollForward>
  </PropertyGroup>
  <ItemGroup>
    <Compile Include=\"Program.cs\" />
    <Compile Include=\"${QUICR_CS_PKG_DIR}/*.cs\" />
  </ItemGroup>
</Project>
"
)

set(QUICR_CS_STAMP ${CMAKE_CURRENT_BINARY_DIR}/qcsclient.stamp)

# `dotnet build -o` pins the output directory so the native library copy
# below has a fixed, predictable target - dotnet's own default output path
# is keyed by configuration/TFM (bin/Debug/net8.0/...), which would need
# a generator-expression-shaped guess for no benefit here.
#
# The copy itself is required, not cosmetic: .NET's default DllImport
# native-library probing checks the app's own base directory first (same
# role LIBRARY_OUTPUT_DIRECTORY plays for the Go/Python examples' own
# language runtimes), and libquicr.{dylib,so} only exists in pkg/ otherwise.
add_custom_command(
    OUTPUT ${QUICR_CS_STAMP}
    COMMAND ${DOTNET_EXECUTABLE} build ${QUICR_CS_APP_DIR}/qcsclient.csproj -o ${QUICR_CS_OUT_DIR}
    COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:quicr_csharp> ${QUICR_CS_OUT_DIR}/
    COMMAND ${CMAKE_COMMAND} -E touch ${QUICR_CS_STAMP}
    WORKING_DIRECTORY ${QUICR_CS_APP_DIR}
    DEPENDS
        quicr_csharp
        $<TARGET_FILE:quicr_csharp>
        ${QUICR_CS_APP_DIR}/Program.cs
        ${QUICR_CS_APP_DIR}/qcsclient.csproj
    COMMENT "Building C# SWIG example (qcsclient)"
    VERBATIM
)

add_custom_target(qcsclient_build ALL DEPENDS ${QUICR_CS_STAMP})
