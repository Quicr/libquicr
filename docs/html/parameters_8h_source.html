221  ParameterList& Add(Type type, const T& value)
222  {
223  const std::uint64_t key = static_cast<std::uint64_t>(type);
224  switch (GetParameterEncoding(type)) {
225  case ParameterEncoding::kByte:
226  if constexpr (ByteParameter<T>) {
227  parameters[key].push_back(static_cast<std::uint8_t>(value));
228  break;
229  } else {
230  throw ProtocolViolationException(
231  "Given parameter type must be u8 (type=" + std::to_string(key) + ")");
232  }
233  case ParameterEncoding::kVarint:
234  if constexpr (requires { UintVar(value); }) {
235  parameters[key] << UintVar(value);
236  break;
237  } else {
238  throw ProtocolViolationException(
239  "Given parameter type must be uvarint(u64) (type=" + std::to_string(key) + ")");
240  }
241  default:
242  if constexpr (std::is_same_v<T, Bytes> || std::is_same_v<T, BytesSpan>) {
243  parameters[key].insert(parameters[key].end(), value.begin(), value.end());
244  } else if constexpr (HasByteStreamOperators<T>) {
245  parameters[key] << value;
246  } else {
247  throw ProtocolViolationException(
248  "Given parameter type must be Location or Bytes (type=" + std::to_string(key) + ")");
249  }
250  break;
251  }
252 
253  return *this;
254  }

310  template<ParameterValueType T>
311  T Get(Type type) const
312  {
313  auto bytes = Find(type);
314  if (bytes.empty()) {
315  return {};
316  }
317 
318  switch (GetParameterEncoding(type)) {
319  case ParameterEncoding::kByte:
320  if constexpr (ByteParameter<T>) {
321  return static_cast<T>(bytes[0]);
322  } else {
323  throw ProtocolViolationException("Given parameter type must be u8 (type=" +
324  std::to_string(static_cast<std::uint64_t>(type)) + ")");
325  }
326  case ParameterEncoding::kVarint:
327  if constexpr (requires(T v) { UintVar(v); }) {
328  return static_cast<T>(UintVar(bytes).Get());
329  } else {
330  throw ProtocolViolationException("Given parameter type must be uvarint(u64) (type=" +
331  std::to_string(static_cast<std::uint64_t>(type)) + ")");
332  }
333  default:
334  if constexpr (std::is_same_v<T, Bytes> || std::is_same_v<T, BytesSpan>) {
335  return Bytes{ bytes.begin(), bytes.end() };
336  } else if constexpr (HasByteStreamOperators<T>) {
337  T result;
338  bytes >> result;
339  return result;
340  } else {
341  throw ProtocolViolationException("Given parameter type must be Location or Bytes (type=" +
342  std::to_string(static_cast<std::uint64_t>(type)) + ")");
343  }
344  }

393  template<typename Type = ParameterType>
394  BytesSpan operator>>(BytesSpan buffer, ParameterList<Type>& msg)
395  {
396  std::uint64_t num = 0;
397  buffer = buffer >> num;
398 
399  std::uint64_t prev_type = 0;
400  std::uint64_t delta = 0;
401  for (std::uint64_t i = 0; i < num; ++i) {
402  buffer = buffer >> delta;
403 
404  const std::uint64_t type = prev_type += delta;
405  Bytes& value = msg.parameters[type];
406 
407  switch (GetParameterEncoding(static_cast<Type>(type))) {
408  case ParameterEncoding::kByte: {
409  std::uint8_t byte = 0;
410  buffer = buffer >> byte;
411  value.push_back(byte);
412  break;
413  }
414  case ParameterEncoding::kVarint: {
415  UintVar uval(buffer);
416  value << uval;
417  buffer = buffer.subspan(uval.size());
418  break;
419  }
420  case ParameterEncoding::kLocation: {
421  Location loc{};
422  buffer = buffer >> loc;
423  value << loc;
424  break;
425  }
426  case ParameterEncoding::kBytes: {
427  buffer = buffer >> value;
428  break;
429  }
430  }
431  }
432 
433  return buffer;
434  }
