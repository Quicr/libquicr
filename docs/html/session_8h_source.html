562  template<typename... Fields>
563  void SendCtrlMsg(const std::shared_ptr<Stream>& stream,
564  bool close_stream,
565  messages::ControlMessageType type,
566  Fields&&... args)
567  {
568  messages::Message msg = messages::Message{}.PrependType(type).ReserveLength();
569 
570  (msg.Append(args), ...);
571 
572  if (stream == nullptr) {
573  throw std::logic_error("Stream cannot be null");
574  }
575 
576  const auto connection = GetConnectionForStream(stream);
577  if (!connection) {
578  throw std::invalid_argument("Stream has no connection");
579  }
580  auto result = quic_transport_->Enqueue(
581  connection, stream, msg.ToBytes(), 0, 2000, { true, close_stream, false, false });
582 
583  if (!result) {
584  // TODO: Propagate better to callers and handle more specifically. Keeping the throw for now.
585  throw TransportException(TransportError::kInvalidStreamId);
586  }
587  }
