35  void AddValue(const uint64_t value)
36  {
37  min = value_count ? std::min(min, value) : value;
38  max = std::max(max, value);
39 
40  value_sum += value;
41  value_count++;
42 
43  if (!value_count)
44  value_count = 1;
45 
46  avg = value_sum / value_count;

57  void Merge(const MinMaxAvg& other)
58  {
59  if (!other.value_count) {
60  return;

63  min = value_count ? std::min(min, other.min) : other.min;

66  value_sum += other.value_sum;
67  value_count += other.value_count;
68 
69  avg = value_sum / value_count;
70  }
