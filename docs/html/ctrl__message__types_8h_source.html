230  struct KeyValuePair
231  {
232  T type;
233  Bytes value;
234 
241  std::size_t Size(const T prev_type) const
242  {
243  std::size_t size = 0;
244  const auto type_value = static_cast<std::uint64_t>(type);
245  const auto prev_type_value = static_cast<std::uint64_t>(prev_type);
246  const auto delta = type_value - prev_type_value;
247  size += UintVar(delta).size();
248 
249  if (type_value % 2 == 0) {
250  // Even types: single varint of value
251  if (value.size() > sizeof(std::uint64_t)) {
252  throw std::invalid_argument("Value too large to encode as uint64_t.");
253  }
254  std::uint64_t val = 0;
255  std::memcpy(&val, value.data(), value.size());
256  size += UintVar(val).size();
257  } else {
258  // Odd types: length + bytes
259  size += UintVar(value.size()).size();
260  size += value.size();
261  }
262  return size;
263  }
264 
271  bool operator==(const KeyValuePair<T>& other) const
272  {
273  if (type != other.type) {
274  return false;
275  }
276 
277  if (static_cast<std::uint64_t>(type) % 2 != 0) {
278  // Odd types are byte equality.
279  return value == other.value;
280  }
281 
282  // Even types are numeric equality.
283  if (value.size() > sizeof(std::uint64_t) || other.value.size() > sizeof(std::uint64_t)) {
284  throw std::invalid_argument("Even KVPs must be <= 8 bytes");
285  }
286 
287  // Compare numeric values.
288  const auto smaller = std::min(value.size(), other.value.size());
289  if (memcmp(value.data(), other.value.data(), smaller) != 0) {
290  return false;
291  }
292 
293  // Are there left over bytes to check?
294  const auto larger = std::max(value.size(), other.value.size());
295  if (larger == smaller) {
296  return true;
297  }
298 
299  // Any remaining bytes could be 0, but nothing else.
300  const auto& longer = (value.size() > other.value.size()) ? value : other.value;
301  const auto remaining = larger - smaller;
302  static constexpr std::uint8_t kZero[sizeof(std::uint64_t)] = { 0 };
303  return memcmp(longer.data() + smaller, kZero, remaining) == 0;
304  }
305  };

344  template<KeyType T>
345  void SerializeKvp(Bytes& buffer, const KeyValuePair<T>& kvp, const T prev_type)
346  {
347  // Delta encode the type.
348  const auto type_value = static_cast<std::uint64_t>(kvp.type);
349  const auto prev_type_value = static_cast<std::uint64_t>(prev_type);
350  const auto delta = type_value - prev_type_value;
351  buffer << UintVar(delta);
352 
353  if (type_value % 2 != 0) {
354  // Odd, encode bytes.
355  buffer << kvp.value;
356  } else {
357  // Even, single varint of value.
358  if (kvp.value.size() > sizeof(std::uint64_t)) {
359  throw std::invalid_argument("Value too large to encode as uint64_t.");
360  }
361  std::uint64_t val = 0;
362  std::memcpy(&val, kvp.value.data(), kvp.value.size());
363  buffer << UintVar(val);
364  }
365  }

375  template<KeyType T>
376  void ParseKvp(BytesSpan& buffer, KeyValuePair<T>& kvp, const T prev_type)
377  {
378  // Delta to absolute.
379  const auto prev_type_value = static_cast<std::uint64_t>(prev_type);
380  std::uint64_t delta;
381  buffer = buffer >> delta;
382  if (delta > std::numeric_limits<std::uint64_t>::max() - prev_type_value) {
383  throw ProtocolViolationException("Delta encoding overflow: prev_type + delta exceeds 2^64-1");
384  }
385  const auto type_value = prev_type_value + delta;
386  kvp.type = static_cast<T>(type_value);
387 
388  if (type_value % 2 != 0) {
389  // Odd, decode bytes.
390  buffer = buffer >> kvp.value;
391  } else {
392  // Even, decode single varint of value.
393  UintVar uvar(buffer);
394  buffer = buffer.subspan(uvar.size());
395  std::uint64_t val(uvar);
396  const auto* p = reinterpret_cast<const std::uint8_t*>(&val);
397  kvp.value.assign(p, p + sizeof(val));
398  }
399  }
