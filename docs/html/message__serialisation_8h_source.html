130  static constexpr std::uint8_t kExtensionsBit = 0x01;
131  static constexpr std::uint8_t kSubgroupIdMask = 0x06;
132  static constexpr std::uint8_t kEndOfGroupBit = 0x08;
133 
135  static constexpr std::uint8_t kTypeBit = 0x10;
136  static constexpr std::uint8_t kDefaultPriorityBit = 0x20;
137  static constexpr std::uint8_t kFirstObjectBit = 0x40;
138 
140  static constexpr std::uint64_t kAllowedBitsMask =
141  kExtensionsBit | kSubgroupIdMask | kEndOfGroupBit | kTypeBit | kDefaultPriorityBit | kFirstObjectBit;
142 
148  explicit constexpr StreamHeaderProperties(const std::uint64_t type)
149  : extensions(type & kExtensionsBit)
150  , subgroup_id_mode(static_cast<SubgroupIdType>((type & kSubgroupIdMask) >> 1))
151  , end_of_group(type & kEndOfGroupBit)
152  , default_priority(type & kDefaultPriorityBit)
153  , first_object(type & kFirstObjectBit)
154  {
155  if (!IsValid(type)) {
156  throw ProtocolViolationException("Invalid stream header type");
157  }
158  }

169  constexpr StreamHeaderProperties(const bool extensions,
170  const SubgroupIdType subgroup_id_mode,
171  const bool end_of_group,
172  const bool default_priority,
173  const bool first_object)
174  : extensions(extensions)
175  , subgroup_id_mode(subgroup_id_mode)
176  , end_of_group(end_of_group)
177  , default_priority(default_priority)
178  , first_object(first_object)
179  {
180  if (subgroup_id_mode == SubgroupIdType::kReserved) {
181  throw ProtocolViolationException("Subgroup ID mode cannot be kReserved");
182  }
183  }
184 
186  constexpr std::uint64_t GetType() const
187  {
188  std::uint64_t type = kTypeBit;
189  if (extensions) {
190  type |= kExtensionsBit;
191  }
192  type |= static_cast<std::uint64_t>(subgroup_id_mode) << 1;
193  if (end_of_group) {
194  type |= kEndOfGroupBit;
195  }
196  if (default_priority) {
197  type |= kDefaultPriorityBit;
198  }
199  if (first_object) {
200  type |= kFirstObjectBit;
201  }
202  return type;
203  }
204 
207  static constexpr bool IsValid(const std::uint64_t type) noexcept
208  {
209  return (type & ~kAllowedBitsMask) == 0 && (type & kTypeBit) != 0 &&
210  (type & kSubgroupIdMask) != kSubgroupIdMask;
211  }
