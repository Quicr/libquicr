137  constexpr void CheckCloseStream(std::uint64_t stream_id, bool is_server, StreamOperation operation)
138  {
139  const bool is_bidir = (stream_id & 0x2) == 0;
140  const bool is_locally_initiated = ((stream_id & 0x1) != 0) == is_server;
141  const bool can_send = is_bidir || is_locally_initiated;
142  const bool can_receive = is_bidir || !is_locally_initiated;
143 
144  switch (operation) {
145  case StreamOperation::kFin:
146  [[fallthrough]];
147  case StreamOperation::kReset:
148  if (can_send) {
149  return;
150  }
151  break;
152  case StreamOperation::kStopSending:
153  if (can_receive) {
154  return;
155  }
156  break;
157  case StreamOperation::kCancel:
158  if (is_bidir) {
159  return;
160  }
161  break;
162  }
163 
164  throw std::invalid_argument("Stream close is invalid for the stream direction");
165  }
