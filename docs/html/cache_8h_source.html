232  void Advance()
233  {
234  const TickType new_ticks =
235  static_cast<uint64_t>(std::chrono::duration_cast<std::chrono::milliseconds>(clock_->Get()).count());
236  const TickType delta = current_ticks_ ? (new_ticks - current_ticks_) / interval_ : 0;
237  current_ticks_ = new_ticks;
238 
239  if (delta == 0) {
240  return;
241  }
242 
243  if (delta >= static_cast<TickType>(total_buckets_)) {
244  Clear();
245  return;
246  }
247 
248  for (TickType i = 0; i < delta; ++i) {
249  auto& bucket = buckets_[(bucket_index_ + i) % total_buckets_];
250  for (const auto& key : bucket) {
251  cache_.erase(key);
252  }
253  bucket.clear();
254  bucket.shrink_to_fit();
255  }
256 
257  bucket_index_ = (bucket_index_ + delta) % total_buckets_;
258  }
